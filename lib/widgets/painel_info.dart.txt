import 'package:flutter/material.dart';

class PainelInfo extends StatelessWidget {
  final bool ativo;
  final double distancia;
  final Duration duracao;

  const PainelInfo({
    super.key,
    required this.ativo,
    required this.distancia,
    required this.duracao,
  });

  String get _duracaoStr {
    final h = duracao.inHours.toString().padLeft(2, '0');
    final m = (duracao.inMinutes % 60).toString().padLeft(2, '0');
    final s = (duracao.inSeconds % 60).toString().padLeft(2, '0');
    return '$h:$m:$s';
  }

  @override
  Widget build(BuildContext context) {
    return Container(
      margin: const EdgeInsets.all(12),
      padding: const EdgeInsets.all(14),
      decoration: BoxDecoration(
        color: const Color(0xFF1E1E2F),
        borderRadius: BorderRadius.circular(12),
        border: ativo ? Border.all(color: Colors.greenAccent, width: 1.5) : null,
      ),
      child: Column(
        crossAxisAlignment: CrossAxisAlignment.start,
        children: [
          Row(
            children: [
              Icon(Icons.circle, color: ativo ? Colors.greenAccent : Colors.grey, size: 10),
              const SizedBox(width: 6),
              Text(
                ativo ? 'Em tempo real' : 'Parado',
                style: TextStyle(color: ativo ? Colors.greenAccent : Colors.grey, fontWeight: FontWeight.w600),
              ),
            ],
          ),
          const SizedBox(height: 8),
          Text(
            'Distância\n${distancia.toStringAsFixed(2)} km • Duração: $_duracaoStr',
            style: const TextStyle(color: Colors.white70, fontSize: 13),
          ),
        ],
      ),
    );
  }
}
